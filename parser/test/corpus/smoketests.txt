# This is a comment
let main = { _ opaque ->
  let one = 1
  let answer = 3.14159
  let string = "hello"

  # The issue:
  # "expect" was in tail position while not being an
  # expression so tree-sitter didn't commit to this
  # being a function declaration at all so it
  # backtracked to "let" and tried to continue
  # to parse it as "identifier" instead
  expect one != answer reporting "Oh, no!"

  # The fix:
  # Actually put an expression in tail position
  answer
}
